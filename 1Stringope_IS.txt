class BitwiseString {
    public static void main(String[] args) {
        String str = "\\Hello World9";

        System.out.println("Original String: " + str);

        // --- AND Operation ---
        System.out.println("\nAfter AND with 127:");
        for (int i = 0; i < str.length(); i++) {
            char result = (char)(str.charAt(i) & 127);
            System.out.print(result);
        }

        // --- OR Operation ---
        System.out.println("\n\nAfter OR with 127:");
        for (int i = 0; i < str.length(); i++) {
            int value = (str.charAt(i) | 127);
            if (value >= 32 && value <= 126) {
                System.out.println(value + " -> " + (char)value);
            } else {
                System.out.println(value + " -> non-printable/control");
            }
        }

        // --- XOR Operation ---
        System.out.println("\nAfter XOR with 127:");
        for (int i = 0; i < str.length(); i++) {
            int value = (str.charAt(i) ^ 127);
            if (value >= 32 && value <= 126) {
                System.out.println(value + " -> " + (char)value);
            } else {
                System.out.println(value + " -> non-printable");
            }
        }
        System.out.println();
    }
}
